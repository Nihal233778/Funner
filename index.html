Use comma‑separated numbers and an integer window ≥ 2.'); return; } if (prices.length < window * 2 - 1) { await ctx.reply(`Not enough data. Need at least ${window * 2 - 1} prices for window ${window}.`); return; } const result = tma(prices, window); let response = `📈 TMA (window=${window})\n`; for (let i = 0; i < prices.length; i++) { const val = result[i]; const display = val !== null ? val.toFixed(4) : '—'; response += `#${i+1} (${prices[i]}) → ${display}\n`; } await ctx.reply(response); } catch (err) { await ctx.reply(`Error: ${err.message}`); } finally { // ⭐ SHOW AD AFTER EVERY BOT RESPONSE try { // Basic usage: await chanify.showAd(ctx.chat.id, ctx.from); // Optional: With country and delay // await chanify.showAd(ctx.chat.id, ctx.from, { // country: "DE", // optional - ISO country code // afterDelay: 1000 // ms before sending the ad // }); } catch (adErr) { console.error('Chanify ad failed:', adErr.message); } } }); // ---------- START BOT ---------- bot.launch() .then(() => console.log('🤖 Bot is running...')) .catch(err => console.error('Launch error:', err)); // Enable graceful stop process.once('SIGINT', () => bot.stop('SIGINT')); process.once('SIGTERM', () => bot.stop('SIGTERM'));
